What's inside

Every layer,
written by hand.

From the 16-bit MBR to the FAT32 driver, every subsystem is original code. Nothing borrowed, nothing abstracted away.

Two-Stage Bootloader

Stage 1 in 512-byte MBR space. Stage 2 sets up GDT, IDT, A20 line, VBE graphics, paging, and loads the kernel from FAT32.

boot.asm · stage2.asm

Memory Manager

Bitmap-based physical page allocator, virtual memory manager with page tables, and a boundary-tag heap with kmalloc / kfree.

pmm · vmm · heap

Full FAT32 Driver

Mount, open, read, write, seek, directory listing, stat, and file creation — a complete filesystem stack over ATA PIO.

fs/fat32

VBE Framebuffer

1024×768 32-bit color graphics mode with pixel drawing, an 8×16 bitmap font, and a full graphical boot splash with progress bar.

vbe.c

Eclipse Shell (esh)

A full shell with line editing, 128-entry history, scripting (if/else, pipes, redirection), variables, aliases, and 30+ built-in commands.

shell.c

.E32 Executable Format

Eclipse32's own binary format with a C runtime, syscall interface, basic libc integration, and interactive stdin support.

ekuecho_pkg

// =============================================================================
// Eclipse32 Kernel — kmain.c  (main branch / VGA text mode)
// Entry point from entry.asm. EAX = boot_info_t*
// =============================================================================
void kmain(boot_info_t *boot_info) {
    // Phase 1: VGA text mode — early debug output
    vga_init();
    vga_clear();
    vga_set_color(VGA_COLOR_CYAN, VGA_COLOR_BLACK);
    vga_puts("Eclipse32 Kernel v0.1\n");

    // Phase 2: CPU structures
    gdt_init();   // GDT + TSS
    idt_init();   // 256-entry IDT
    pic_remap(0x20, 0x28);
    pit_init(1000);  // 1000 Hz timer tick
    syscall_init();

    asm volatile("sti");  // enable interrupts

    // Phase 3: Memory management
    uint32_t mem_start = (uint32_t)&kernel_end;
    pmm_init(mem_start, 32*1024*1024 - mem_start);
    vmm_init(boot_info);
    heap_init(0xC0000000, 4*1024*1024);  // 4MB at 0xC0000000

    // Phase 4: Drivers
    keyboard_init();
    ata_init();
    // VBE disabled — VGA text mode branch

    // Phase 5: Filesystem + launch shell
    fat32_mount(0);
    initramfs_start();   // → ESH shell, never returns
    kpanic("InitRAMFS returned unexpectedly!");
}

// Line editor — handles arrow keys, Ctrl+A/E/K/L/C, Home/End/Delete
static int readline(char *buf, int maxlen) {
    int len = 0, cur = 0, hist_idx = hist_count;
    while (1) {
        key_event_t e = keyboard_wait_event();
        if (e.keycode == KEY_ENTER) break;
        if (e.keycode == KEY_BACKSPACE && cur > 0) {
            memmove(buf+cur-1, buf+cur, len-cur);
            buf[--len] = 0; --cur;
            // redraw line...
        }
        // ... arrow keys, Ctrl sequences, history navigation
    }
    buf[len] = '\0';
    return len;
}

// apps/sdk/e32_syscall.h — .E32 app syscall interface
// Apps link against this to call kernel services via INT 0x80

#define E32_MAGIC   0xEC320001  // .E32 file magic

// Syscall wrappers callable from C apps
static inline void e32_exit(int code) {
    asm volatile("int $0x80"
        :: "a"(SYS_EXIT), "b"(code));
}

static inline int e32_write(int fd, const void *buf, size_t n) {
    int ret;
    asm volatile("int $0x80"
        : "=a"(ret)
        : "a"(SYS_WRITE), "b"(fd), "c"(buf), "d"(n));
    return ret;
}

static inline int e32_read(int fd, void *buf, size_t n) {
    int ret;
    asm volatile("int $0x80"
        : "=a"(ret)
        : "a"(SYS_READ), "b"(fd), "c"(buf), "d"(n));
    return ret;
}
